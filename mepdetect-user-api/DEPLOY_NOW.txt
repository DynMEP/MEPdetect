📋 DEPLOYMENT COMMANDS
═══════════════════════════════════════════════════════════

# 1. Copy files to server (if not already there)
scp -r /mnt/user-data/outputs/mepdetect-user-management root@your-server:/opt/

# 2. SSH to server
ssh root@your-server

# 3. Run deployment
cd /opt/mepdetect-user-management
chmod +x quick-deploy.sh
sudo ./quick-deploy.sh

# 4. Create admin user (when prompted)
# Email: admin@mepdetect.com
# Password: (your choice)
